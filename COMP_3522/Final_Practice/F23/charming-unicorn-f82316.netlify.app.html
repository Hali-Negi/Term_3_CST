from abc import ABC, abstractmethod

class Switchable(ABC):
    @abstractmethod
    def turn_on(self):
        pass

    @abstractmethod
    def turn_off(self):
        pass

class SwitchController:
    def __init__(self, device):
        self.device = device

    def turn_on(self):
        self.device.turn_on()

    def turn_off(self):
        self.device.turn_off()

class Light(Switchable):
    def turn_on(self):
        print("Light is on")

    def turn_off(self):
        print("Light is off")

class EventListener:
    def __init__(self):
        self.handlers = {}

    def add_event_handler(self, event, callback):
        if event not in self.handlers:
            self.handlers[event] = []
        self.handlers[event].append(callback)

    def remove_event_handler(self, event, callback):
        if event in self.handlers and callback in self.handlers[event]:
            self.handlers[event].remove(callback)

        if event in self.handlers:

class LoanApprovalSystem:
    def __init__(self):
        self.handlers = []

    def add_authority(self, handler):
        self.handlers.append(handler)

    def remove_authority(self, handler):
        if handler in self.handlers:
            self.handlers.remove(handler)
            print(f"Authority {handler.__class__.__name__} removed.")
        else:
            print("Authority not found.")

    def process_request(self, loan):
        approvals = [handler.can_approve(loan) for handler in self.handlers]
        if any(approvals):
            print("Loan approved by at least one authority.")
        else:
