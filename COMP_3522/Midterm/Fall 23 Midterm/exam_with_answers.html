class Account:
    def __init__(self, account_number, balance):
        self.account_number = account_number
        self.balance = balance

    def deposit(self, amount):

        return self.balance

import json

def analyze_json_log(log_file):
    try:
        with open(log_file, "r") as file:
            log_entries = [json.loads(line) for line in file if line.strip()]  # Load JSON from each line

            total_entries = len(log_entries)
            error_count = 0
            warning_count = 0
            info_count = 0

            log_level_counts = {"error": 0, "warning": 0}

            for entry in log_entries:
                level = entry.get('level', '').lower()
                if level in log_level_counts:
                    log_level_counts[level] += 1
            return total_entries, error_count, warning_count, info_count
    except FileNotFoundError:
        print(f"The log file '{log_file}' does not exist.")
